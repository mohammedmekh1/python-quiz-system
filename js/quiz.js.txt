
# إنشاء ملف quiz.js الرئيسي
js_content = '''/**
 * Python Quiz System - Main Application
 * Professional Quiz Application with SurveyJS
 */

class QuizApp {
    constructor() {
        this.survey = null;
        this.examData = null;
        this.studentData = {
            name: '',
            email: '',
            startTime: null,
            endTime: null
        };
        this.timer = null;
        this.timeRemaining = 0;
        this.n8nEndpoint = 'https://n8n.srv1332143.hstgr.cloud/webhook';
        
        this.init();
    }

    init() {
        this.parseURLParams();
        this.loadExam();
    }

    parseURLParams() {
        const urlParams = new URLSearchParams(window.location.search);
        this.examId = urlParams.get('exam') || urlParams.get('exam_id');
        this.token = urlParams.get('token');

        if (!this.examId || !this.token) {
            this.showError('رابط الاختبار غير صحيح. يرجى التأكد من الرابط المقدم.');
            return false;
        }
        return true;
    }

    async loadExam() {
        try {
            const response = await fetch(`${this.n8nEndpoint}/get-survey?exam=${this.examId}&token=${this.token}`, {
                method: 'GET',
                headers: {
                    'Content-Type': 'application/json',
                }
            });

            if (!response.ok) {
                throw new Error('Failed to load exam');
            }

            this.examData = await response.json();
            
            if (this.examData.error) {
                throw new Error(this.examData.error);
            }

            this.showStartForm();
        } catch (error) {
            console.error('Error loading exam:', error);
            this.showError('تعذر تحميل الاختبار. قد يكون الرابط منتهي الصلاحية أو غير صحيح.');
        }
    }

    showStartForm() {
        document.getElementById('loading-state').classList.add('hidden');
        document.getElementById('start-form').classList.remove('hidden');
        
        // Update exam info
        document.getElementById('exam-title').textContent = this.examData.title || 'اختبار بايثون';
        document.getElementById('exam-duration').textContent = 
            `${this.examData.durationMin || 15} دقيقة`;

        // Setup form
        document.getElementById('student-form').addEventListener('submit', (e) => {
            e.preventDefault();
            this.startQuiz();
        });
    }

    startQuiz() {
        // Get student data
        this.studentData.name = document.getElementById('student-name').value.trim();
        this.studentData.email = document.getElementById('student-email').value.trim();
        this.studentData.startTime = new Date();

        if (!this.studentData.name || !this.studentData.email) {
            this.showToast('يرجى إدخال جميع البيانات المطلوبة', 'error');
            return;
        }

        // Hide start form, show quiz
        document.getElementById('start-form').classList.add('hidden');
        document.getElementById('quiz-container').classList.remove('hidden');
        document.getElementById('timer-display').classList.remove('hidden');

        // Initialize timer
        this.startTimer(this.examData.durationMin || 15);

        // Create and render survey
        this.createSurvey();
    }

    startTimer(minutes) {
        this.timeRemaining = minutes * 60;
        
        this.updateTimerDisplay();
        
        this.timer = setInterval(() => {
            this.timeRemaining--;
            this.updateTimerDisplay();
            
            if (this.timeRemaining <= 0) {
                this.autoSubmit();
            }
        }, 1000);
    }

    updateTimerDisplay() {
        const minutes = Math.floor(this.timeRemaining / 60);
        const seconds = this.timeRemaining % 60;
        const display = `${String(minutes).padStart(2, '0')}:${String(seconds).padStart(2, '0')}`;
        
        document.getElementById('timer-value').textContent = display;
        
        // Warning colors
        const timerDisplay = document.getElementById('timer-display');
        if (this.timeRemaining <= 60) {
            timerDisplay.classList.add('timer-warning', 'text-red-600', 'bg-red-50');
            timerDisplay.classList.remove('text-gray-700', 'bg-gray-100');
        } else if (this.timeRemaining <= 300) {
            timerDisplay.classList.add('text-orange-600', 'bg-orange-50');
            timerDisplay.classList.remove('text-gray-700', 'bg-gray-100');
        }
    }

    createSurvey() {
        const surveyJson = this.examData;
        
        // Configure SurveyJS
        Survey.StylesManager.applyTheme("defaultV2");
        
        this.survey = new Survey.Model(surveyJson);
        
        // RTL Support
        this.survey.locale = 'ar';
        
        // Progress tracking
        this.survey.onProgressText.add((sender, options) => {
            const progress = Math.round(sender.getProgress() * 100);
            document.getElementById('progress-bar').style.width = `${progress}%`;
            document.getElementById('progress-text').textContent = `${progress}%`;
        });

        // Question numbering
        this.survey.onTextMarkdown.add((sender, options) => {
            if (options.name === 'question') {
                options.html = options.text;
            }
        });

        // Completion handler
        this.survey.onComplete.add((sender) => {
            this.submitExam(sender.data);
        });

        // Render survey
        this.survey.render(document.getElementById("surveyElement"));
    }

    autoSubmit() {
        clearInterval(this.timer);
        this.showToast('انتهى الوقت! جاري تسليم الإجابات تلقائياً...', 'warning');
        
        if (this.survey) {
            this.survey.completeLastPage();
        }
    }

    async submitExam(answers) {
        clearInterval(this.timer);
        this.studentData.endTime = new Date();
        
        const timeSpent = Math.floor((this.studentData.endTime - this.studentData.startTime) / 1000);
        
        const submissionData = {
            exam: this.examId,
            token: this.token,
            studentName: this.studentData.name,
            studentEmail: this.studentData.email,
            answers: answers,
            timeSpent: timeSpent,
            finishedAt: this.studentData.endTime.toISOString()
        };

        try {
            // Show loading
            document.getElementById('quiz-container').classList.add('hidden');
            document.getElementById('loading-state').classList.remove('hidden');
            document.querySelector('#loading-state p').textContent = 'جاري تصحيح الإجابات...';

            const response = await fetch(`${this.n8nEndpoint}/submit-exam`, {
                method: 'POST',
                headers: {
                    'Content-Type': 'application/json',
                },
                body: JSON.stringify(submissionData)
            });

            if (!response.ok) {
                throw new Error('Submission failed');
            }

            const result = await response.json();
            this.showResults(result);
            
        } catch (error) {
            console.error('Error submitting exam:', error);
            this.showError('حدث خطأ أثناء تسليم الاختبار. يرجى المحاولة مرة أخرى.');
        }
    }

    showResults(result) {
        document.getElementById('loading-state').classList.add('hidden');
        document.getElementById('results-container').classList.remove('hidden');
        document.getElementById('timer-display').classList.add('hidden');

        const percentage = parseFloat(result.results?.percentage || 0);
        const passed = result.results?.passed || percentage >= 60;
        
        // Update score circle
        const degree = (percentage / 100) * 360;
        document.getElementById('score-circle').style.setProperty('--score-degree', `${degree}deg`);
        document.getElementById('score-percentage').textContent = `${percentage}%`;
        
        // Update colors based on score
        const scoreColor = passed ? '#10b981' : '#ef4444';
        document.getElementById('score-circle').style.background = 
            `conic-gradient(${scoreColor} ${degree}deg, #e2e8f0 0deg)`;
        document.getElementById('score-percentage').style.color = scoreColor;

        // Update header
        if (passed) {
            document.getElementById('result-icon').textContent = '🎉';
            document.getElementById('result-title').textContent = 'تهانينا! لقد نجحت';
            document.getElementById('result-message').textContent = 'أداء ممتاز، استمر في التقدم!';
            this.celebrate();
        } else {
            document.getElementById('result-icon').textContent = '💪';
            document.getElementById('result-title').textContent = 'لا تيأس!';
            document.getElementById('result-message').textContent = 'يمكنك إعادة المحاولة وتحسين نتيجتك';
        }

        // Update stats
        document.getElementById('stat-score').textContent = result.results?.totalScore || 0;
        document.getElementById('stat-max').textContent = result.results?.maxScore || 0;
        document.getElementById('stat-correct').textContent = 
            `${result.results?.correctCount || 0}/${result.results?.totalQuestions || 0}`;
        document.getElementById('stat-time').textContent = 
            Math.floor((result.timeSpent || 0) / 60);

        // Show detailed answers
        this.renderDetailedResults(result.detailedResults || []);
    }

    renderDetailedResults(detailedResults) {
        const container = document.getElementById('answers-list');
        container.innerHTML = '';

        detailedResults.forEach((item, index) => {
            const isCorrect = item.isCorrect;
            const statusClass = isCorrect ? 'bg-green-50 border-green-200' : 'bg-red-50 border-red-200';
            const icon = isCorrect ? '✅' : '❌';
            
            const div = document.createElement('div');
            div.className = `result-card rounded-xl p-4 border ${statusClass}`;
            div.innerHTML = `
                <div class="flex items-start gap-3">
                    <span class="text-2xl">${icon}</span>
                    <div class="flex-1">
                        <h4 class="font-semibold text-gray-800 mb-1">السؤال ${index + 1}</h4>
                        <p class="text-gray-600 text-sm mb-2">${item.question}</p>
                        <div class="flex gap-4 text-sm">
                            <span class="text-gray-500">
                                <i class="fas fa-user-edit mr-1"></i>
                                إجابتك: ${item.studentAnswer || 'لم يتم الإجابة'}
                            </span>
                            ${!isCorrect ? `
                                <span class="text-green-600">
                                    <i class="fas fa-check mr-1"></i>
                                    الصحيحة: ${item.correctAnswer}
                                </span>
                            ` : ''}
                        </div>
                        <div class="mt-2 text-sm font-semibold ${isCorrect ? 'text-green-600' : 'text-red-600'}">
                            الدرجة: ${item.score}/${item.maxScore}
                        </div>
                    </div>
                </div>
            `;
            container.appendChild(div);
        });
    }

    celebrate() {
        const duration = 3000;
        const animationEnd = Date.now() + duration;
        const defaults = { startVelocity: 30, spread: 360, ticks: 60, zIndex: 0 };

        const randomInRange = (min, max) => Math.random() * (max - min) + min;

        const interval = setInterval(() => {
            const timeLeft = animationEnd - Date.now();

            if (timeLeft <= 0) {
                return clearInterval(interval);
            }

            const particleCount = 50 * (timeLeft / duration);
            confetti(Object.assign({}, defaults, { 
                particleCount, 
                origin: { x: randomInRange(0.1, 0.3), y: Math.random() - 0.2 } 
            }));
            confetti(Object.assign({}, defaults, { 
                particleCount, 
                origin: { x: randomInRange(0.7, 0.9), y: Math.random() - 0.2 } 
            }));
        }, 250);
    }

    showError(message) {
        document.getElementById('loading-state').classList.add('hidden');
        document.getElementById('error-state').classList.remove('hidden');
        document.getElementById('error-message').textContent = message;
    }

    showToast(message, type = 'info') {
        const container = document.getElementById('toast-container');
        const toast = document.createElement('div');
        
        const colors = {
            success: 'bg-green-500',
            error: 'bg-red-500',
            warning: 'bg-orange-500',
            info: 'bg-blue-500'
        };

        toast.className = `${colors[type]} text-white px-6 py-3 rounded-xl shadow-lg transform transition-all duration-300 translate-y-10 opacity-0`;
        toast.innerHTML = `
            <div class="flex items-center gap-2">
                <i class="fas ${type === 'success' ? 'fa-check' : type === 'error' ? 'fa-times' : 'fa-info-circle'}"></i>
                <span>${message}</span>
            </div>
        `;

        container.appendChild(toast);

        // Animate in
        setTimeout(() => {
            toast.classList.remove('translate-y-10', 'opacity-0');
        }, 100);

        // Remove after 3 seconds
        setTimeout(() => {
            toast.classList.add('translate-y-10', 'opacity-0');
            setTimeout(() => toast.remove(), 300);
        }, 3000);
    }
}

// Utility functions
function toggleTheme() {
    document.body.classList.toggle('dark-theme');
}

function downloadCertificate() {
    alert('سيتم إضافة ميزة تحميل الشهادة قريباً!');
}

// Initialize app when DOM is ready
document.addEventListener('DOMContentLoaded', () => {
    window.quizApp = new QuizApp();
});

// Prevent leaving page during exam
window.addEventListener('beforeunload', (e) => {
    if (window.quizApp && window.quizApp.survey && !window.quizApp.survey.isCompleted) {
        e.preventDefault();
        e.returnValue = '';
    }
});
